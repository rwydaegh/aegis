\documentclass[border=5mm]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}
\usepackage{amsmath,amssymb}

\begin{document}

\tdplotsetmaincoords{70}{120}

\begin{tikzpicture}[tdplot_main_coords, scale=2.2]

% =====================================================
% PARAMETERS
% =====================================================
\pgfmathsetmacro{\thetaI}{40}

\pgfmathsetmacro{\kx}{sin(\thetaI)}
\pgfmathsetmacro{\ky}{0}
\pgfmathsetmacro{\kz}{-cos(\thetaI)}

\pgfmathsetmacro{\karrowlen}{1.6}
\pgfmathsetmacro{\kstartx}{-\karrowlen*\kx}
\pgfmathsetmacro{\kstarty}{-\karrowlen*\ky}
\pgfmathsetmacro{\kstartz}{-\karrowlen*\kz}

% e_s = (0,-1,0),  e_p = (cos40,0,sin40)
\pgfmathsetmacro{\esx}{0}
\pgfmathsetmacro{\esy}{-1}
\pgfmathsetmacro{\esz}{0}
\pgfmathsetmacro{\epx}{cos(\thetaI)}
\pgfmathsetmacro{\epy}{0}
\pgfmathsetmacro{\epz}{sin(\thetaI)}

% Refraction: |n~| = 4.84, thetaT ~ 7.6°
\pgfmathsetmacro{\thetaT}{asin(sin(\thetaI)/4.84)}
% e'_p = (cos(thetaT), 0, sin(thetaT))  [nearly tangential]
\pgfmathsetmacro{\eprx}{cos(\thetaT)}
\pgfmathsetmacro{\epry}{0}
\pgfmathsetmacro{\eprz}{sin(\thetaT)}

% =====================================================
% SURFACE PLANE
% =====================================================
\pgfmathsetmacro{\planeS}{1.5}
\fill[gray!15, draw=black!40, thick]
  (-\planeS,-\planeS,0) -- (\planeS,-\planeS,0) --
  (\planeS,\planeS,0)   -- (-\planeS,\planeS,0) -- cycle;

% =====================================================
% INCIDENCE PLANE — two colours: blue above Sigma, tan below
% =====================================================
% Below surface (tissue side)
\fill[orange!8, draw=none, opacity=0.55]
  (-1.5,0,-0.85) -- (1.5,0,-0.85) -- (1.5,0,0) -- (-1.5,0,0) -- cycle;
% Above surface (air side)
\fill[blue!10, draw=none, opacity=0.5]
  (-1.5,0,0) -- (1.5,0,0) -- (1.5,0,1.5) -- (-1.5,0,1.5) -- cycle;
% Outer border
\draw[blue!30, very thin]
  (-1.5,0,-0.85) -- (1.5,0,-0.85) -- (1.5,0,1.5) -- (-1.5,0,1.5) -- cycle;
% Intersection line at z=0 (where incidence plane meets Sigma)
\draw[blue!60!black, semithick]
  (-1.5,0,0) -- (1.5,0,0);

% =====================================================
% -n̂ REFERENCE LINE (very thin gray, unlabelled) — baseline for theta_t arc
% =====================================================
\draw[gray!40, thin]
  (0,0,0) -- (0, 0, -0.9);

% =====================================================
% REFRACTED RAY k'_n (into tissue, z < 0)
% =====================================================
\pgfmathsetmacro{\krlen}{0.85}
\draw[-latex, thick, black!50, dashed]
  (0,0,0) -- ({\krlen*sin(\thetaT)}, 0, {-\krlen*cos(\thetaT)});
\node[black!60, font=\small] at ({0.75*sin(\thetaT)}, 0.08, {-0.75*cos(\thetaT)-0.13})
  {$\hat{\mathbf{k}}'_n$};

% theta_t arc: from -n̂ direction (straight down) to k'_n
\pgfmathsetmacro{\arcRT}{0.32}
\draw[thin, black!45] plot[domain=0:\thetaT, samples=25, smooth]
  ({\arcRT*sin(\x)}, 0, {-\arcRT*cos(\x)});
\pgfmathsetmacro{\labelAngleT}{\thetaT/2}
\node[black!55, font=\footnotesize] at
  ({0.48*sin(\labelAngleT)}, 0.05, {-0.48*cos(\labelAngleT)}) {$\theta_t$};

% =====================================================
% NORMAL VECTOR
% =====================================================
\draw[-latex, very thick, black]
  (0,0,0) -- (0,0,1.3) node[above] {$\hat{\mathbf{n}}$};

% =====================================================
% INCIDENT WAVE VECTOR k_n
% =====================================================
\pgfmathsetmacro{\kstop}{0.35}
\pgfmathsetmacro{\ktipx}{\kstartx+\kstop*(-\kstartx)}
\pgfmathsetmacro{\ktipy}{\kstarty+\kstop*(-\kstarty)}
\pgfmathsetmacro{\ktipz}{\kstartz+\kstop*(-\kstartz)}
\draw[-latex, very thick, red!70!black]
  (\kstartx,\kstarty,\kstartz) -- (\ktipx,\ktipy,\ktipz);
\draw[dashed, thick, red!70!black]
  (\ktipx,\ktipy,\ktipz) -- (0,0,0);
\node[red!70!black, right] at
  ({0.55*\kstartx},{0.55*\kstarty},{0.55*\kstartz}) {$\hat{\mathbf{k}}_n$};

% =====================================================
% INCIDENCE POINT
% =====================================================
\fill[black] (0,0,0) circle (1.2pt);

% =====================================================
% e_{s,n}
% =====================================================
\draw[-latex, very thick, green!50!black]
  (0,0,0) -- ({1.1*\esx},{1.1*\esy},{1.1*\esz})
  node[below left] {$\hat{e}_{s,n}$};

% =====================================================
% e_{p,n}  (incident TM direction — large normal component)
% =====================================================
\draw[-latex, very thick, orange!80!black]
  (0,0,0) -- ({1.1*\epx},{1.1*\epy},{1.1*\epz})
  node[right] {$\hat{e}_{p,n}$};

% =====================================================
% e'_{p,n}  (refracted TM direction — nearly tangential)
% label placed below-left of arrow tip to avoid overlap
% =====================================================
\pgfmathsetmacro{\eprlen}{1.0}
\draw[-latex, very thick, cyan!50!blue]
  (0,0,0) -- ({\eprlen*\eprx},{\eprlen*\epry},{\eprlen*\eprz});
\node[cyan!40!blue, font=\small] at ({\eprlen*\eprx+0.0},{\eprlen*\epry-0.18},{\eprlen*\eprz-0.16})
  {$\hat{e}'_{p,n}$};

% =====================================================
% psi_n
% =====================================================
\pgfmathsetmacro{\psComp}{0.7}
\pgfmathsetmacro{\ppComp}{0.6}
\pgfmathsetmacro{\psix}{\psComp*\esx+\ppComp*\epx}
\pgfmathsetmacro{\psiy}{\psComp*\esy+\ppComp*\epy}
\pgfmathsetmacro{\psiz}{\psComp*\esz+\ppComp*\epz}
\pgfmathsetmacro{\projSx}{\psComp*\esx}
\pgfmathsetmacro{\projSy}{\psComp*\esy}
\pgfmathsetmacro{\projSz}{\psComp*\esz}
\pgfmathsetmacro{\projPx}{\ppComp*\epx}
\pgfmathsetmacro{\projPy}{\ppComp*\epy}
\pgfmathsetmacro{\projPz}{\ppComp*\epz}
\draw[dashed,semithick,black!40]
  (\psix,\psiy,\psiz)--(\projSx,\projSy,\projSz);
\draw[dashed,semithick,black!40]
  (\psix,\psiy,\psiz)--(\projPx,\projPy,\projPz);
\draw[-latex, very thick, purple!70!black]
  (0,0,0)--(\psix,\psiy,\psiz) node[left]{$\boldsymbol{\psi}_n$};

% =====================================================
% theta_n arc (above surface, between n_hat and -k_hat)
% =====================================================
\pgfmathsetmacro{\arcR}{0.55}
\draw[thin] plot[domain=0:\thetaI,samples=30,smooth]
  ({-\arcR*sin(\x)},0,{\arcR*cos(\x)});
\pgfmathsetmacro{\labelAngle}{\thetaI/2}
\node at ({-0.72*sin(\labelAngle)},0,{0.72*cos(\labelAngle)}) {$\theta_n$};

% dashed extension of -k_hat to make theta_n clear
\draw[thin,dashed,red!40!black]
  (0,0,0)--({-0.8*sin(\thetaI)},0,{0.8*cos(\thetaI)});

% =====================================================
% LABELS
% =====================================================
\node[blue!50!black] at (-0.6,0,1.4) {\footnotesize incidence plane};
\node[black!60] at (1.2,-1.2,0) {$\Sigma$};

\end{tikzpicture}
\end{document}
