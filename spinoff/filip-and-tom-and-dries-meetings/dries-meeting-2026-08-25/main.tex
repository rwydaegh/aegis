\begin{frame}{Surface law}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[height=4.35cm]{fig_geometry.pdf}
    \end{column}
    \begin{column}{0.44\textwidth}
      \vspace{8pt}
      {\small\begin{equation*}
        \begin{aligned}
        \Sab(\rr)={}&{\color{colsinc}\Sinc(\rr)}
        {\color{coltransmission}\Teff(\rr)}\\[-1pt]
        &{\color{colgeometry}[\nhat(\rr)\!\cdot\!(-\khat)]_+}
        \Vis(\rr,\khat)
        \end{aligned}
      \end{equation*}}
      \vspace{4pt}
      \pause
      \begin{itemize}\small\setlength\itemsep{5pt}
        \item $[\nhat\cdot(-\khat)]_+$: projected area factor
        \item $\Vis$: 0 if the body shadows the point, 1 otherwise
        \item $\Teff$: from the Fresnel equations at the surface
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{2pt}
  \pause
  \begin{limitbox}[Validity]
    \centering\small
    Electrically large, locally smooth surface. Optically thick lossy tissue. Given illumination. Curvature parameter: $1/(kR)$.
  \end{limitbox}
\end{frame}

\begin{frame}{Tissue response}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \centering
      \includegraphics[width=0.98\linewidth]{apd_angle_panel_T.pdf}
    \end{column}
    \begin{column}{0.45\textwidth}
      \vspace{2pt}
      {\small Polarization-aware base model:}
      \begin{equation*}
        \Teff=|e_s|^2T_s+|e_p|^2T_p.
      \end{equation*}
      \pause
      \begin{itemize}\footnotesize\setlength\itemsep{3pt}
        \item Fresnel coefficients give the transmitted fraction per path
        \item For unpolarized, high-loss tissue, $\Teff$ is close to $\Tzero$
        \item A planar layer stack can improve total-power estimates
      \end{itemize}
      \vspace{2pt}
      \begin{limitbox}[Limits of $T_0$]
        \scriptsize $T_0$ is a high-index, unpolarized approximation. Less accurate near grazing incidence.
      \end{limitbox}
    \end{column}
  \end{columns}
  \vspace{1pt}
\end{frame}

\begin{frame}{Coherence}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \centering
      \includegraphics[height=5.8cm]{channel_exposure_flow.pdf}
    \end{column}
    \begin{column}{0.38\textwidth}
      \vspace{0pt}
      {\small\begin{equation*}
        P_{\mathrm{abs}}=\xx^H\QQ\xx,
        \qquad
        \QQ=\int_{\Sigma}\Gtilde^H\Gtilde\,\diff A.
      \end{equation*}}
      \pause
      \begin{itemize}\scriptsize\setlength\itemsep{2pt}
        \item $\QQ$ is Hermitian positive semidefinite
        \item Random path phases average out cross terms
        \item Once built, $\QQ$ is reusable across precoders
        \item $\Gtilde$ uses a mean-depth approximation ($<0.08\%$ discrepancy in tested band)
      \end{itemize}
      \vspace{2pt}
      \pause
      \begin{limitbox}[Evidence limit]
        \scriptsize Derived and tested numerically. Cross-term accuracy has no independent full-wave validation.
      \end{limitbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Fresnel response}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=0.95\linewidth,height=5.2cm,keepaspectratio]{incidence_plane_fig.pdf}
    \end{column}
    \begin{column}{0.44\textwidth}
      \small
      With $\mu=\cos\theta_i$, complex refractive index $\ntilde$ and
      $\xi=\sqrt{\ntilde^2-1+\mu^2}$,
      \begin{align*}
        r_s&=\frac{\mu-\xi}{\mu+\xi},
        &T_s&=1-|r_s|^2,\\
        r_p&=\frac{\ntilde^2\mu-\xi}{\ntilde^2\mu+\xi},
        &T_p&=1-|r_p|^2,\\
        \Teff&=|e_s|^2T_s+|e_p|^2T_p.&&
      \end{align*}
      \begin{limitbox}
        \footnotesize Local planar boundary solution only. Full-body scattering is not included.
      \end{limitbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Validity limits}
  \small
  \begin{tabularx}{\linewidth}{@{}p{32mm} >{\raggedright\arraybackslash}X >{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Claim} & \textbf{Scope} & \textbf{Escalation} \\
    \midrule
    Local APD map & 6 to 100 GHz under given illumination, with scale- and geometry-dependent error & layered resonance, source-body coupling, strong diffraction \\
    Whole-body absorption & roughly 1 to 100 GHz with layered total-power response & electrically small body or low-loss penetration \\
    Device pre-screening & 10 to 60 GHz radiating near field, after benchmark & incident field changed by the body \\
    Differentiability & wherever paths and shadows stay fixed & shadow or path birth and death \\
    Coherent operator & PSD by construction, tested numerically & off-diagonal accuracy not validated \\
    \bottomrule
  \end{tabularx}
  \vspace{8pt}
  \begin{resultbox}\centering\small
    Validity depends on observable, geometry and source model together. Frequency alone is not enough.
  \end{resultbox}
\end{frame}

\begin{frame}{Application filter}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{evidencebox}[Eligibility]
        \begin{enumerate}\footnotesize\setlength\itemsep{3pt}
          \item Observable is local or integrated dissipated power
          \item Object is electrically large and locally smooth
          \item Material is sufficiently opaque and lossy
          \item Incident field is known without strong body coupling
          \item Repeated evaluations justify the reduction
        \end{enumerate}
      \end{evidencebox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{limitbox}[Exclusions]
        \begin{itemize}\footnotesize\setlength\itemsep{5pt}
          \item Antenna detuning or general S-parameters
          \item Total scattering or certifiable RCS
          \item Edge-dominated signatures
          \item Resonant thin or low-loss structures
          \item Reactive near-field source coupling
        \end{itemize}
      \end{limitbox}
    \end{column}
  \end{columns}
  \vspace{9pt}
  \begin{questionbox}\centering\small
    Each candidate needs a separate prior-art search and benchmark.
  \end{questionbox}
\end{frame}
