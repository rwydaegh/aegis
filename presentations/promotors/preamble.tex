% ===============================================================
% HEADLINE — 3-tab navigation bar
% ===============================================================
% Sections are numbered 1, 2, 3, ... by \section{} commands.
% We map: section 2 = Part I, section 3 = Part II, section 4 = Part III.
% (Section 1 = Front matter, which has no tab highlight.)

\newcommand{\navtab}[3]{%
  % #1 = hyperlink target label
  % #2 = display text
  % #3 = section number that activates this tab
  \hyperlink{#1}{%
    \ifnum\value{section}=#3%
      \colorbox{activetab}{\color{white}\small\strut\;\textbf{#2}\;}%
    \else%
      \colorbox{inactivetab}{\color{white}\small\strut\;\textsf{#2}\;}%
    \fi%
  }%
}

\setbeamertemplate{headline}{%
  \vskip3pt%
  \hbox to \paperwidth{%
    \hfill%
    \navtab{sec:part1}{Part I: mmWave Framework}{2}%
    \hskip6pt%
    \navtab{sec:part2}{Part II: Generalisations}{3}%
    \hskip6pt%
    \navtab{sec:part3}{Part III: Coherent MIMO}{4}%
    \hfill%
  }%
  \vskip3pt%
  {\color{navyblue}\hrule height 0.4pt}%
}
