% SLIDE 2: Back-of-envelope calculation
% SAY: "EMF compliance for 5G deployments requires computing
%       absorbed power density. The standard approach, FDTD,
%       is too slow. Quick back of the envelope..."
% ----------------------------------------------------------
\begin{frame}{Back-of-envelope calculation}

  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \onslide<1->{%
      EMF compliance assessment requires computing absorbed
      power density. The standard approach is FDTD.

      \vspace{6pt}
      Per dimension: $\SI{2}{\metre}\,/\,\SI{0.2}{\milli\metre} = 10^{4}$

      \vspace{4pt}
      Volumetric FDTD mesh (3D):
      \[
        (10^{4})^3 \;=\; {\color{alertred}10^{12} \text{ cells}}
      \]
      One plane wave, one phantom, one frequency: \textbf{weeks}.
      }

      \vspace{8pt}
      \onslide<2->{%
      Our formulation works on the body surface (2D):
      \[
        {\sim}\SI{1.7}{\metre^2}\text{ at mm resolution}
        \;\approx\; {\color{resultgreen}10^{6} \text{ elements}}
      \]
      The depth direction is solved analytically (Fresnel).
      }
    \end{column}

    \begin{column}{0.36\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=0.75]
          \draw[thick, navyblue] (0,0) -- (0,4);
          \draw[<->] (0.3,0) -- (0.3,4)
            node[midway,right] {\footnotesize $\SI{2}{\metre}$};
          \draw[thick, alertred, fill=alertred!20]
            (-0.04,3.96) rectangle (0.12,4);
          \draw[<->] (0.5,3.96) -- (0.5,4)
            node[midway,right] {\footnotesize $\delta \approx \SI{0.2}{\milli\metre}$};
          \node at (0,-0.5) {\footnotesize body};
          \node[alertred] at (0.8,4.3) {\footnotesize skin depth};
          \node[gray,rotate=90] at (-0.5,2) {\tiny not to scale};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \vspace{4pt}
  \onslide<2->{%
  \begin{resultbox}
    \centering
    $10^{12} \;\to\; 10^{6}$: six orders of magnitude.
  \end{resultbox}
  }
\end{frame}

\begin{frame}{Exact absorption law}

  \vspace{-2pt}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \onslide<1->{%
      \vspace{6pt}
      {\large $\displaystyle
        \Sab(\rr) \;=\;
        {\color{colsinc}\Sinc}
        \onslide<2->{\;\cdot\; {\color{coltransmission}\Teff(\rr)}}
        \onslide<3->{\;\cdot\; {\color{colgeometry}\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]}}
      $}
      }

      \vspace{12pt}
      \onslide<1->{%
      \begin{itemize}
        \item {\color{colsinc}$\Sinc$}: incident power density.\\
              Set by the environment.
      \end{itemize}
      }

      \vspace{4pt}
      \onslide<2->{%
      \begin{itemize}
        \item {\color{coltransmission}$\Teff$}: Fresnel transmission.\\
              Depends on angle, polarisation, tissue, frequency.
      \end{itemize}
      }

      \vspace{4pt}
      \onslide<3->{%
      \begin{itemize}
        \item {\color{colgeometry}$\ReLU$}: visibility gate (exact).\\
              Back-facing points absorb nothing.
      \end{itemize}
      }

      \vspace{8pt}
      \onslide<4->{%
      {\small The material factor depends on four variables.\\
       We will see that it simplifies to a single number.}
      }
    \end{column}

    \begin{column}{0.43\textwidth}

\begin{frame}{The geometric absorption law}

  \onslide<1->{%
  \begin{center}
    \begin{resultbox}[Geometric absorption law]
      \vspace{-4pt}
      {\large\[
        \Sab(\rr) \;\approx\;
        {\color{colsinc}\Sinc}
        \;\cdot\;
        {\color{coltransmission}T_0}
        \;\cdot\;
        {\color{colgeometry}\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]}
      \]}
      \vspace{-6pt}
    \end{resultbox}
  \end{center}
  }

  \vspace{6pt}
  \onslide<2->{%
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      The pseudo-Brewster effect makes the
      Fresnel transmission nearly angle-independent.

      \vspace{6pt}
      $T_0$ for skin at \SI{28}{\giga\hertz}: \textbf{0.54}.

      \vspace{4pt}
      {\small
      \begin{tabular}{lcr}
        \toprule
        Frequency & $T_0$ (skin) & Error \\
        \midrule
        \SI{10}{\giga\hertz} & 0.47 & $\le 8\,\%$ \\
        \SI{28}{\giga\hertz} & 0.54 & $\le 5.6\,\%$ \\
        \SI{60}{\giga\hertz} & 0.57 & $\le 4.2\,\%$ \\
        \bottomrule
      \end{tabular}
      }
    \end{column}

    \begin{column}{0.48\textwidth}
      All spatial variation from body shape.\\[6pt]
      Holds for $|\ntilde| > 2.5$ (Azzam, 2015).\\[4pt]
      Biological tissue: $|\ntilde| \approx 3$--6.
    \end{column}
  \end{columns}
  }
\end{frame}

% SLIDE 12: Exposure operator and ECBF
% SAY: "Integrate over the body surface: you get Q.
%       Q captures the body's complete absorption response
%       to any precoder. It has a closed-form solution."
% ----------------------------------------------------------
\begin{frame}{Exposure-constrained beamforming}

  \onslide<1->{%
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{resultbox}[Coherent absorption law]
        \vspace{-6pt}
        \[
          \Sab(\rr) = \bigl\|\Gtilde(\rr)\,\xx\bigr\|^2
        \]
        \vspace{-8pt}
      \end{resultbox}

      \vspace{4pt}
      \begin{resultbox}[Exposure operator]
        \vspace{-6pt}
        \[
          P_{\mathrm{abs}} = \xx^H \QQ\, \xx
        \]
        \vspace{-10pt}
        \[
          \QQ = \int_\Sigma \Gtilde^H \Gtilde\,\diff A
          \;\in\; \Complex^{M \times M}
        \]
        \vspace{-8pt}
      \end{resultbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      $\QQ$ captures the body's complete absorption\\
      response to any precoder.

      \vspace{6pt}
      \onslide<2->{%
      Maximise signal subject to absorption limit:
      \[
        \max_{\xx} \;|\hh^\top \xx|^2
        \;\;\text{s.t.}\;\;
        \xx^H \QQ\, \xx \le P_{\mathrm{abs}}^{\max}
      \]

      Closed-form solution:
      \[
        \xx^\star = \sqrt{P}\;
        \frac{(\lambda\QQ + \nu\II)^{-1}\hh^*}
             {\|(\lambda\QQ + \nu\II)^{-1}\hh^*\|}
      \]
      {\footnotesize
        $\lambda\QQ$ suppresses high-absorption directions.\\
        $\lambda = 0$: MRT. \quad $\lambda \gg \nu$: body-avoiding.
      }
      }
    \end{column}
  \end{columns}
  }
\end{frame}
