\begin{frame}{Two physical reductions behind the dosimetry law}
  \vspace{2pt}
  Conventional dosimetry solves Maxwell's equations over the whole body volume
  (FDTD, ${\sim}10^{12}$ cells, hours to days). Two physical facts reduce that
  to a single evaluation on the body \emph{surface}:
  \vspace{6pt}
  \begin{center}
    {\large $\displaystyle
      \Sab(\rr) \;=\;
      {\color{colsinc}\Sinc}\;\cdot\;
      {\color{coltransmission}T_0}\;\cdot\;
      {\color{colgeometry}\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]}
    $}
  \end{center}
  \vspace{6pt}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \centering {\color{colsinc}$\Sinc$}\\[2pt]
      {\footnotesize incident power density, set by the environment}
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering {\color{coltransmission}$T_0$ : one scalar}\\[2pt]
      {\footnotesize the material factor, from pseudo-Brewster compensation}
    \end{column}
    \begin{column}{0.33\textwidth}
      \centering {\color{colgeometry}pure geometry}\\[2pt]
      {\footnotesize only the outer shape matters, from surface confinement}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \begin{resultbox}\centering
    Reduction 1 replaces the material with one scalar. Reduction 2 replaces the body
    volume with its surface.
  \end{resultbox}
\end{frame}


\begin{frame}{1.\ Pseudo-Brewster compensation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \vspace{2pt}
      A wave entering tissue splits into TE and TM polarizations, each with its
      own Fresnel transmission.
      \begin{itemize}\setlength\itemsep{2pt}
        \item As the angle grows, {\color{colsinc}$T_s$ (TE) falls} while
              {\color{coltransmission}$T_p$ (TM) rises} to the
              \emph{pseudo-Brewster} peak (${\approx}0.95$)
        \item Moving oppositely, the unpolarized average
              $\Tavg=\tfrac12(T_s+T_p)$ stays nearly flat
        \item True for any high-index medium ($|\ntilde|\gtrsim 2.5$), and tissue at
              mmWave is $|\ntilde|\in[3,6]$
      \end{itemize}
      \vspace{3pt}
      \begin{resultbox}\centering
        \footnotesize Replace $\Tavg(\theta)$ by one scalar:
        $T_0 = 0.539$ (skin, \SI{28}{\giga\hertz}).
      \end{resultbox}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=0.96\linewidth]{apd_angle_panel_T.pdf}\\[1pt]
      {\scriptsize $T_s$ falls and $T_p$ peaks, but $\Tavg$ stays within
      \SI{5.6}{\percent} of $T_0$ out to $75^\circ$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{1.\ Pseudo-Brewster: validity and contribution}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=0.93\linewidth]{R_of_f.pdf}\\[1pt]
      {\scriptsize Sphere ratio $R(f)=T_0/\Tbar$. $R<1$ means $T_0$
      \emph{underestimates} absorbed power, so the result is conservative for
      compliance. Within $\pm\SI{4}{\percent}$ across the band, with the crossover
      at \SI{40.4}{\giga\hertz}.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{honestbox}[Prior art and contribution]
        {\small\raggedright
        Azzam (2015) proved this near-constant transmission for high-index
        \emph{optical} substrates ($|\ntilde|>3.73$). Kodera, Bamba, Flintoft and
        Zhang \emph{measured} it in tissue above \SI{6}{\giga\hertz}.\\[5pt]
        The contribution here is the derivation for biological tissue from Fresnel
        theory, used as the material factor in a body-surface dosimetry system. The
        scalar is enabling physics, not the claimed invention.}
      \end{honestbox}
    \end{column}
  \end{columns}
\end{frame}


\begin{frame}{2.\ Surface confinement}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \vspace{2pt}
      Above a few GHz the field penetrates only a thin skin of the tissue before it
      is fully absorbed.
      \vspace{3pt}

      \begin{resultbox}\centering
        \footnotesize A ${\sim}10^{12}$-cell volume mesh becomes a
        ${\sim}10^{4}$-triangle surface integral.
      \end{resultbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2.\ Surface confinement: validity and contribution}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \vspace{2pt}
      The surface picture is valid above ${\sim}\SI{6}{\giga\hertz}$, the band where
      the regulatory metric is already surface absorbed power density, not
      whole-body SAR.
      \vspace{6pt}

      Below ${\sim}\SI{6}{\giga\hertz}$ the field reaches past the subcutaneous fat,
      Fabry--P\'erot resonances appear, and the local surface map loses pointwise
      meaning. Total absorbed power stays valid there through a layered
      transmission $\Tbar$.
    \end{column}
    \begin{column}{0.5\textwidth}
      \vspace{2pt}
      \begin{honestbox}[Prior art and contribution]
        {\footnotesize
        A sub-millimetre skin depth at mmWave is textbook electromagnetics. The
        contribution here is the resulting method: volumetric dosimetry reduces to a
        surface integral over the body's outer shape, the incidence factor
        $\ReLU[\nhat\cdot(-\khat)]$ gates the lit side against the shadow, and the
        valid frequency window is bounded.}
      \end{honestbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Both reductions give the law}
  \vspace{4pt}
  \begin{center}
    {\large $\displaystyle
      \Sab(\rr) \;=\;
      {\color{colsinc}\Sinc}\;\cdot\;
      \underbrace{{\color{coltransmission}T_0}}_{\text{reduction 1}}\;\cdot\;
      \underbrace{{\color{colgeometry}\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]}}_{\text{reduction 2}}
    $}
  \end{center}
  \vspace{6pt}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\setlength\itemsep{3pt}
        \item {\color{coltransmission}Pseudo-Brewster} reduces the angle- and
              polarization-dependent material to one scalar $T_0$
        \item {\color{colgeometry}Surface confinement} reduces the body volume to a
              surface, with absorption gated by the incidence cosine
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\setlength\itemsep{3pt}
        \item The same law also covers near-field point sources and coherent MIMO
        \item It is differentiable end to end, so antenna and device parameters
              can be optimized by gradient descent
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{4pt}
  \begin{resultbox}\centering
    These two are the enabling physics. The independent claims in the IDF
    (the spatial law and its near-field forms, the exposure operator $\QQ$, the
    differentiable optimization, and the systems) are built on top of them.
  \end{resultbox}
\end{frame}

\begin{frame}{Appendix: exact Fresnel coefficients}
  For a planar interface between free space and a lossy medium with complex
  refractive index $\ntilde=\sqrt{\varepsilon_r - i\sigma/(\omega\varepsilon_0)}$,
  and normal wave-vector component $\xi=\sqrt{\ntilde^2-1+\mu^2}$ with
  $\mu=\cos\theta_i$:
  \begin{equation*}
    r_s = \frac{\mu-\xi}{\mu+\xi}, \qquad
    r_p = \frac{\ntilde^2\mu-\xi}{\ntilde^2\mu+\xi},\qquad
    T_{s,p} = 1 - |r_{s,p}|^2 .
  \end{equation*}
  At normal incidence both reduce to
  \[
    T_0 = \frac{4\,\RE(\ntilde)}{|1+\ntilde|^2}.
  \]
  For skin at \SI{28}{\giga\hertz}: $\varepsilon_r=16.55$, $\sigma=\SI{25.8}{\siemens\per\metre}$,
  $\ntilde=4.49-1.79i$, $T_0=0.539$.
\end{frame}

\begin{frame}{Appendix: pseudo-Brewster across tissues (\SI{28}{\giga\hertz})}
  \centering\small
  \begin{tabular}{lcccc}
    \toprule
    Tissue & $|\ntilde|$ & $T_0$
      & \makecell{pointwise dev.\\$\Tavg/T_0-1$, $[0,75^\circ]$}
      & \makecell{absorbed-power\\(flux-weighted) dev.} \\
    \midrule
    Skin   & 4.8 & 0.54 & $+5.7\%$  & $-1.2\%$ \\
    Muscle & 5.7 & 0.48 & $+12.6\%$ & $-2.9\%$ \\
    Water  & 6.9 & 0.41 & $+22.3\%$ & $-5.1\%$ \\
    Fat    & 2.6 & 0.78 & $-17.4\%$ & $+4.6\%$ \\
    \bottomrule
  \end{tabular}
  \vspace{8pt}

  {\footnotesize Computed from the IT'IS tissue model. The pointwise $\Tavg(\theta)$
  does rise (or, for fat, fall) at grazing angles, more so for the higher-index
  tissues. But absorbed power scales as $T(\theta)\cos\theta$, so the
  \emph{absorbed-power-weighted} deviation stays within ${\sim}5\%$ for every tissue.
  That is the quantity the direction-averaged $R(f)$ reflects, and why the single
  scalar is accurate for absorbed power even where the raw transmission is not flat.}
\end{frame}
