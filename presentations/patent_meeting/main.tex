% ---------------------------------------------------------- 1. PROBLEM
\begin{frame}{The problem}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \phvid[3.4cm]{PhantomScene: zoom whole body $\to$ surface triangle $\to$ skin depth $\delta$}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item The body is about \SI{1}{\metre} across, but the skin depth at \SI{28}{\giga\hertz} is only $\delta\!\approx\!\SI{0.3}{\milli\metre}$
        \item FDTD resolves that with $\sim 10^{12}$ cells and weeks of GPU per run
      \end{itemize}
      \vspace{8pt}
      \begin{itemize}
        \item Above \SI{6}{\giga\hertz} the limit is absorbed power density on the skin, not whole-body SAR
        \item Compliance is mandatory for every 5G/6G device and base station, at every product cycle
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{3pt}
  \begin{resultbox}\centering
    The regulatory `gold standard' is computationally intractable at product and network scale.
  \end{resultbox}
\end{frame}

% ---------------------------------------------------------- A1
\begin{frame}{Framework A: the essential steps}
  It is a surface computation, not a volume simulation:
  \begin{enumerate}
    \item Build the body surface mesh and its triangle normals $\nhat$
    \item Take one tissue transmission coefficient $T_0$ per frequency
    \item For each source, set the incident power density $\Sinc$ and the direction $\khat$
    \item Apply the exact incidence gate $\ReLU[\nhat\cdot(-\khat)]$, so back-facing points absorb nothing
    \item Spatially average over \SI{4}{\cm\squared} / \SI{1}{\cm\squared} and check ICNIRP compliance
  \end{enumerate}
  \vspace{2pt}
  \begin{tcolorbox}[colback=lightblue!40,colframe=medblue,arc=2pt,boxrule=0.5pt,left=5pt,right=5pt,top=3pt,bottom=3pt]
    {\footnotesize
    \textbf{Essential}: steps 1--4. \textbf{Variable}: transmission model ($T_0/\Tavg/\Tbar$), mesh detail, fidelity level. \textbf{Deployed}: the far-field path.}
  \end{tcolorbox}
\end{frame}

% ---------------------------------------------------------- A2 pseudo-Brewster
\begin{frame}{Framework A: material collapses to one number}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\linewidth]{apd_angle_panel_T.pdf}
    \end{column}
    \begin{column}{0.4\textwidth}
      \vspace{8pt}
      As the angle changes, $T_s$ falls and $T_p$ rises to the pseudo-Brewster peak. The unpolarized average stays flat at one scalar $T_0$.

      \vspace{10pt}
      {\footnotesize Azzam proved this constancy for high-index optical substrates. The contribution here is the derivation for tissue and the integration into a body-surface system, not the scalar itself.}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------- A3 geometry
\begin{frame}{Framework A: it is all geometry}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \centering
      \censorphantom[0.48\linewidth]{sab_phantom_visible.pdf}{0.310,0.880}{0.560,0.915}{0.260,0.440}{0.480,0.490}\hfill%
      \censorphantom[0.48\linewidth]{eta_phantom_front.pdf}{0.250,0.870}{0.460,0.905}{0.260,0.440}{0.480,0.490}
    \end{column}
    \begin{column}{0.52\textwidth}
      Once the material is a scalar, all spatial structure comes from body shape:
      \begin{itemize}
        \item the lit side absorbs at the cosine rate, the shadow side absorbs nothing
        \item self-shadowing $\eta$ is the ambient-occlusion primitive of computer graphics
      \end{itemize}
      \vspace{2pt}
      \begin{resultbox}[Whole body (direction-averaged)]
        \vspace{-6pt}
        \[ \langle P_{\mathrm{abs}}\rangle = \Sinc\, T_0\, \Aab/4 \]
      \end{resultbox}
      {\footnotesize This is a generalized Cauchy identity, so whole-body compliance reduces to three precomputed scalars.}
    \end{column}
  \end{columns}
\end{frame}

  \begin{center}
    \only<1>{\includegraphics[width=0.85\linewidth, trim=0 30 0 30, clip]{channel_block_panel_a.png}}%

  \end{center}
  \onslide<3->{{\small Signal sees a scalar $\hh^\top\xx$. Exposure sees a vector field $\Gtilde(\rr)\,\xx$. The two lead to different optimization targets.}}
\end{frame}

% ---------------------------------------------------------- B2 exposure operator
\begin{frame}{Framework B: the exposure operator}
  \begin{resultbox}[Closed form, no FDTD calibration]
    \vspace{-6pt}
    \[
      \Sab(\rr)=\|\Gtilde(\rr)\,\xx\|^2,\qquad
      P_{\mathrm{abs}}=\xx^H\QQ\,\xx,\qquad
      \QQ=\int_\Sigma \Gtilde^H\Gtilde\,\diff A
    \]
  \end{resultbox}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item $\QQ$ is Hermitian PSD, derived from Fresnel and geometry
        \item the exposure-constrained beamformer has a closed form: $\xx^\star\propto(\lambda\QQ+\nu\II)^{-1}\hh^*$
        \item it replaces the FDTD-calibrated SAR matrix used in prior MIMO work
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[height=3.4cm, trim=0 0 0 146.6, clip]{ecbf_pareto/ecbf_pareto.pdf}\\[1pt]
      {\scriptsize ECBF frontier, UE behind the phantom}
    \end{column}
  \end{columns}
  \vspace{1pt}
  {\footnotesize\color{warnamber} The theorems are proved and implemented in the engine. The paper is in progress, and the quantitative numbers are still preliminary.}
\end{frame}

% ---------------------------------------------------------- B3 near-field device
\begin{frame}{Framework B: near-field device (prototype)}
  \begin{resultbox}[The same law, with a point source]
    \vspace{-6pt}
    \[
      \Sab(\rr) = \frac{P_t\,G(\khat(\rr))}{4\pi\,d(\rr)^2}\; T_0\;
      \ReLU\!\bigl[\nhat(\rr)\cdot(-\khat(\rr))\bigr]
    \]
  \end{resultbox}
  \vspace{1pt}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      The plane wave becomes a point source at range $d(\rr)=|\rr-\rr_s|$, with local direction $\khat(\rr)=(\rr-\rr_s)/d(\rr)$. Only $\Sinc$ and $\khat$ turn position-dependent, and the visibility gate is unchanged.

      \vspace{4pt}
      {\footnotesize At mmWave the reactive near field ends within \SI{1.7}{\milli\metre} at \SI{28}{\giga\hertz}, so any device beyond contact sits in the locally plane-wave regime.}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item it is a research prototype on ray-traced scenes, not a shipped tool
        \item it proves device pre-compliance is feasible
        \item it is the main commercial opportunity, and the focus of the IOF proposal
        \item it may need its own claim cluster
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
