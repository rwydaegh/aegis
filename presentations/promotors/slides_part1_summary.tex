% SLIDE 6: Exact absorption law
% SAY: "Start from Poynting's theorem at a locally flat surface
%       point. The wave decomposes into TE and TM; Fresnel
%       gives the transmission for each. The ReLU gate is exact:
%       back-facing points receive zero power."

\begin{frame}{Exact absorption law}
\hypertarget{sec:part1}{}
\hypertarget{sum-p1-exact-law}{}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Three factors determine local absorption:

      \begin{itemize}
        \item Illumination: incident power density $\Sinc$
        \item Material: effective Fresnel transmission $\Teff(\rr)$
        \item Geometry: visibility gate $\ReLU\!\bigl(\nhat\cdot(-\khat)\bigr)$
      \end{itemize}

      \vspace{6pt}
      \begin{resultbox}[Exact absorption law]
        \vspace{-6pt}
        \[
          \Sab(\rr) = \Sinc \;\Teff(\rr) \;\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]
        \]
      \end{resultbox}

      \vspace{4pt}
      {\footnotesize
        $\Teff = |e_s|^2 T_s(\theta) + |e_p|^2 T_p(\theta)$: Fresnel-weighted average.\\
        $\ReLU$ gate is exact: back-facing points absorb nothing.\\
        \gotodetail{app-p1-fresnel}{Fresnel derivation}\;
        \gotodetail{app-p1-exact}{Exact law details}
      }
    \end{column}

    \begin{column}{0.4\textwidth}
      \begin{center}
        \includegraphics[width=\textwidth]{incidence_plane_fig.pdf}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 7: Pseudo-Brewster compensation
% SAY: "This is THE key physics insight. For biological tissue,
%       TE transmission decreases with angle, TM increases.
%       Their average stays near T_0. Azzam proved this in optics
%       for |n| > 2.5. Skin has |n| = 4.84. The material factor
%       collapses to a single number."
% ----------------------------------------------------------
\begin{frame}{Pseudo-Brewster compensation}
\hypertarget{sum-p1-brewster}{}

  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      $T_s$ decreases with angle; $T_p$ increases.\\
      Average $\Tavg(\theta)$ stays near $T_0$.

      \vspace{6pt}
      {\small
      \begin{tabular}{lccr}
        \toprule
        Tissue & $|\ntilde|$ & $T_0$ & $\Tavg/T_0$ var. \\
        \midrule
        Skin   & 4.84 & 0.54 & $\le 5.6\,\%$ \\
        Muscle & 5.62 & 0.48 & $\le 4.8\,\%$ \\
        Fat    & 2.05 & 0.77 & $\le 8.2\,\%$ \\
        Water  & 6.62 & 0.45 & $\le 3.9\,\%$ \\
        \bottomrule
      \end{tabular}
      }

      \vspace{6pt}
      \begin{resultbox}[Geometric absorption law]
        \vspace{-6pt}
        \[
          \Sab(\rr) \approx \Sinc \; T_0 \;\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]
        \]
      \end{resultbox}
    \end{column}

    \begin{column}{0.52\textwidth}
      \begin{center}
        \includegraphics[width=\textwidth]{fresnel_curves.png}
      \end{center}
      {\footnotesize Azzam (2015): holds for $|\ntilde| > 2.5$.\\
        Biological tissue: $|\ntilde| \approx 3$--6. \checkmark\\
        \gotodetail{app-p1-pB}{Pseudo-Brewster details}}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 8: Geometric consequences
% SAY: "Once the material collapses to T_0, everything is
%       geometry. Total absorbed power is T_0 times the shadow
%       area. Direction-average gives Cauchy's formula from 1841.
%       The self-shadowing correction eta is identical to ambient
%       occlusion in computer graphics."
% ----------------------------------------------------------
\begin{frame}{Geometric consequences}
\hypertarget{sum-p1-geometry}{}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Total absorbed power: $P_{\mathrm{abs}} = \Sinc \, T_0 \, \Aperp(\khat)$\quad ($\Aperp$ = shadow area)

      \vspace{1pt}
      Self-shadowing on non-convex bodies:
      {\small $\displaystyle\eta(\rr) = \tfrac{1}{\pi}\!\int_{S^2}
        \ReLU[\nhat\!\cdot\!(-\khat)]\; O(\rr,\khat)\,\diff\Omega$}

      \vspace{1pt}
      $\equiv$ ambient occlusion (GPU graphics);
      Thelonious: $\Aab/A \approx 0.87$.

      \vspace{1pt}
      \begin{resultbox}[Generalised Cauchy formula]
        \vspace{-8pt}
        \[
          \langle P_{\mathrm{abs}} \rangle = \Sinc \, T_0 \, \frac{\Aab}{4}
        \]
        \vspace{-8pt}
      \end{resultbox}
      {\footnotesize Classical Cauchy ($\langle\Aperp\rangle = A/4$ for convex) is special case $\eta\equiv1$.\\
       \gotodetail{app-p1-geometry}{Cauchy proof \& ambient occlusion}}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{center}
        \includegraphics[width=\textwidth,height=0.65\textheight,keepaspectratio]{eta_3d_phantom.png}
      \end{center}
      \vspace{-4pt}
      {\footnotesize Exposure fraction $\eta(\rr)$ on Thelonious.\\
       Red = exposed, blue = self-shadowed.}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 9: Absorption directivity
% SAY: "The directivity D(k) encodes how the body's shape
%       modulates absorption. It ranges from 0.8 to 2.0 on
%       Thelonious. Smooth on the sphere: 49 SH coefficients
%       capture it to 1% RMS."
% ----------------------------------------------------------
\begin{frame}{Absorption directivity}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Factored result:}
      \[
        P_{\mathrm{abs}}(\khat) = \Sinc\, T_0\, \frac{\Aab}{4}\, D(\khat)
      \]

      \vspace{4pt}
      Separation of concerns:
      \begin{itemize}
        \item EM environment: $\Sinc$
        \item Tissue: $T_0$ (one number)
        \item Body geometry: $\Aab$ and $D(\khat)$, both precomputed
      \end{itemize}

      \vspace{6pt}
      $D(\khat)$ is smooth on $S^2$:\\
      SH expansion with $L \le 6$ (49 coeff.)\\
      gives RMS error $< 1\,\%$.
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{center}
        \includegraphics[width=\textwidth]{body_directivity_thelonious.png}
      \end{center}
      {\footnotesize Absorption directivity $D(\khat)$,\\
       Mollweide projection. Range: 0.8--2.0.}
    \end{column}
  \end{columns}
\end{frame}

% SLIDE 10: Computational framework
% SAY: "For N paths from a ray tracer, the spatial dosimetry
%       map is a single matrix multiply with a ReLU. This is
%       literally a single-hidden-layer neural network.
%       The analogy is structural. The chain is differentiable."
% ----------------------------------------------------------
\begin{frame}{Computational framework}

  \textbf{Multi-source environment:} $N$ paths with directions $\KK$, power densities $\ssv$:

  \begin{resultbox}[Dosimetry as a ReLU neural network]
    \vspace{-6pt}
    \[
      \mathbf{S}_{\mathrm{ab}} = T_0 \;\ReLU(\NN\KK^\top)\; \ssv
    \]
  \end{resultbox}

  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item $\NN$: surface normals (body mesh)
        \item $\KK$: path directions (ray tracer)
        \item $\ssv$: path power densities (ray tracer)
        \item ReLU: visibility gating (exact)
      \end{itemize}

      \vspace{4pt}
      \textbf{Differentiable}: gradients of SAR w.r.t.\ antenna positions propagate by backpropagation through any ray-tracing engine.
    \end{column}

    \begin{column}{0.4\textwidth}
      \textbf{Point source} ($d > 3\lambda$):
      \[
        P_{\mathrm{abs}} = P_t \, T_0 \, \frac{\Omega_{\mathrm{body}}}{4\pi}
      \]
      {\footnotesize Emissive view-factor formula\\
       of radiative heat transfer,\\
       with $T_0$ replacing emissivity.}
    \end{column}
  \end{columns}
\end{frame}

      \begin{itemize}
        \item Tests Fresnel + geometric optics jointly
