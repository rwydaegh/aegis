% A-P3-1: Setup and inherited assumptions
% ----------------------------------------------------------
\begin{frame}[label=app-p3-setup]{Setup and inherited assumptions}

  \textbf{Antenna array and precoding:}
  \begin{itemize}
    \item $M$ antenna elements, precoding vector $\xx \in \Complex^M$ with $\|\xx\|^2 = P$
    \item $N$ propagation paths from ray tracer or channel model
    \item Each path $n$ carries polarisation-amplitude vector $\bm{\psi}_n \in \Complex^3$
  \end{itemize}

  \vspace{6pt}
  \textbf{Inherited from Parts I--II:}
  \begin{itemize}
    \item Locally flat surface at each body-surface point $\rr$
    \item Opaque body ($|\ntilde| > 2.5$, skin-depth regime)
    \item Fresnel transmission at each surface point
    \item Depth-integrated Ohmic loss defines $\Sab(\rr)$
  \end{itemize}

  \vspace{6pt}
  \textbf{What changes in Part III:}
  \begin{itemize}
    \item \textcolor{alertred}{Coherent summation of complex amplitudes} (not powers)
    \item Cross-terms $\bm{\psi}_n^H \bm{\psi}_{n'} \, x_{j(n)} x_{j(n')}^*$ can no longer be ignored
    \item Must retain exact complex Fresnel coefficients $t_s, t_p$ (not $\sqrt{T_0}$)
  \end{itemize}

  \vspace{4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% ----------------------------------------------------------
% A-P3-2: Propagation paths
% ----------------------------------------------------------
\begin{frame}[label=app-p3-paths]{Propagation paths}

  {\small Each path $n = 1,\ldots,N$ is characterised by a triplet:}
  \vspace{-2pt}
  \begin{resultbox}[Path descriptor]
    \vspace{-4pt}
    \[
      \bigl(\, j(n),\;\; \khat_n,\;\; \bm{\psi}_n \,\bigr)
      \qquad
      \begin{cases}
        j(n) \in \{1,\ldots,M\} & \text{feeding antenna element}\\
        \khat_n \in \Reals^3,\; \|\khat_n\|=1 & \text{direction of arrival}\\
        \bm{\psi}_n \in \Complex^3 & \text{polarisation-amplitude vector}
      \end{cases}
    \]
  \end{resultbox}
  \vspace{0pt}
  {\small $\bm{\psi}_n$ encodes all propagation interactions:}
  \begin{itemize}\setlength\itemsep{0pt}
    \item {\small Complex amplitude (including path loss and phase)}
    \item {\small Polarisation state (elliptical in general)}
    \item {\small Reflection, diffraction, scattering along the path}
  \end{itemize}
  \vspace{-2pt}
  {\small \textbf{At body surface point $\rr$:}}
  \begin{itemize}\setlength\itemsep{0pt}
    \item {\small Local incidence angle: $\theta_n(\rr) = \arccos\bigl(\nhat(\rr) \cdot (-\khat_n)\bigr)$}
    \item {\small Wave decomposes into TE/TM relative to local incidence plane}
    \item {\small TE/TM unit vectors $\mathbf{e}_{s,n}(\rr)$ and $\mathbf{e}_{p,n}(\rr)$ depend on $\khat_n$ and $\nhat(\rr)$}
  \end{itemize}
  \vspace{-4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% A-P3-3: Field channel matrix
% ----------------------------------------------------------
\begin{frame}[label=app-p3-G]{Field channel matrix}
  {\small \textbf{Field at $\rr$ due to all $M$ elements:}}
  \vspace{-2pt}
  \[
    \EE(\rr) = \sum_{n=1}^{N} \bm{\psi}_n \, x_{j(n)}
  \]
  \vspace{-2pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Signal channel} (scalar at UE):
      \[
        \hh \in \Complex^M, \quad y = \hh^\top \xx
      \]
      Each element of $\hh$: scalar sum of path contributions arriving at UE.
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{Exposure channel} (vector at body):
      \[
        \Gtilde(\rr) \in \Complex^{3 \times M}
      \]
      Column $j$ of $\Gtilde(\rr)$:
      \[
        [\Gtilde(\rr)]_j = \sum_{\substack{n:\, j(n)=j}} F_n(\rr)\,\bm{\psi}_n
      \]
    \end{column}
  \end{columns}

  \vspace{-2pt}
  \begin{resultbox}[Key structural difference]
    \vspace{-4pt}
    \begin{center}
      Signal sums scalars $\;\longrightarrow\; \hh^\top\xx \in \Complex$
      \qquad\qquad
      Exposure sums vectors $\;\longrightarrow\; \Gtilde(\rr)\,\xx \in \Complex^3$
    \end{center}
  \end{resultbox}
  \vspace{-2pt}
  {\footnotesize $F_n(\rr)$: Fresnel transmission operator (rank-2 projector), detailed in slide \hyperlink{app-p3-fresnel-op}{\beamergotobutton{A-P3-5}}.}
  \vspace{-4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

\begin{frame}{Signal vs exposure: key differences}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Signal branch:}
      \begin{enumerate}
        \item Propagation paths $\bm{\psi}_n$
        \item Scalar collapse at UE antenna
        \item Received signal: $y = \hh^\top \xx$
      \end{enumerate}

      \vspace{4pt}
      {\footnotesize UE antenna selects one polarisation component $\rightarrow$ scalar per path.}
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{Exposure branch:}
      \begin{enumerate}
        \item Same propagation paths $\bm{\psi}_n$
        \item Fresnel filter $F_n(\rr)$ at body surface
        \item Vector field: $\|\Gtilde(\rr)\,\xx\|^2$
      \end{enumerate}

      \vspace{4pt}
      {\footnotesize Fresnel filter retains polarisation and angular dependence $\rightarrow$ $3 \times M$ matrix per surface point.}
    \end{column}
  \end{columns}

  \vspace{8pt}
  Both branches share the same propagation paths and precoding vector $\xx$.\\
  The structural difference is why signal focus $\neq$ body hotspot.

% ----------------------------------------------------------
% A-P3-5: Fresnel transmission operator
% ----------------------------------------------------------
\begin{frame}[label=app-p3-fresnel-op]{Fresnel transmission operator}

  {\small For path $n$ at surface point $\rr$:}
  \vspace{-1pt}
  \begin{resultbox}[Fresnel operator]
    \vspace{-4pt}
    \[
      F_n(\rr) = \Bigl(\,
        t_{s,n}\;\mathbf{e}_{s,n}\,\mathbf{e}_{s,n}^\top
        \;+\;
        t_{p,n}\;\mathbf{e}_{p,n}\,\mathbf{e}_{p,n}^\top
      \,\Bigr)\;
      H\!\bigl(\mu_n(\rr)\bigr)
    \]
  \end{resultbox}
  \vspace{-2pt}
  {\footnotesize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Components:}
      \begin{itemize}\setlength\itemsep{0pt}
        \item $t_{s,n}$, $t_{p,n}$: complex Fresnel transmission coeff.
        \item $\mathbf{e}_{s,n}$, $\mathbf{e}_{p,n}$: TE/TM unit vectors
        \item $H(\mu_n)$: Heaviside step (visibility gate)
        \item $\mu_n(\rr) = \nhat(\rr) \cdot (-\khat_n)$
      \end{itemize}
      \vspace{-2pt}
      $F_n$: rank-2 if front-facing ($\mu_n > 0$), rank-0 if back-facing.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Why exact $t_s, t_p$ matter:}
      \begin{itemize}\setlength\itemsep{0pt}
        \item Cross-terms depend on phase of $t_s, t_p$
        \item $\sqrt{T_0}$ replacement destroys phases
        \item Cross-term error: unbounded
        \item Self-terms: $|t_s|^2 = T_s$, $|t_p|^2 = T_p$
      \end{itemize}
      \vspace{-2pt}
      Incoherent (Part I): only $|t|^2$ matters.
      Coherent (Part III): full complex $t$ required.
    \end{column}
  \end{columns}
  }
  \vspace{-4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% A-P3-6: Coherent depth integral and Approximation 1
% ----------------------------------------------------------
\begin{frame}[label=app-p3-depth]{Coherent depth integral and Approximation 1}

  {\small
  \textbf{Absorbed power density} = Ohmic loss integrated over depth $z$:
  \[
    \Sab(\rr) = \int_0^\infty \frac{\sigma}{2}\,\bigl|\EE_t(\rr,z)\bigr|^2\,\diff z
  \]
  \vspace{-2pt}
  Expand $|\EE_t|^2$: self-terms + cross-terms weighted by depth coupling:
  \[
    \Lambda_{nn'}(\rr) = \int_0^\infty \frac{\sigma}{2}\,
    e^{-\alpha_n z}\,e^{-\alpha_{n'}^* z}\,\diff z
    \qquad\text{(complex decay rates $\alpha_n$, $\alpha_{n'}$)}
  \]
  }
  \vspace{-1pt}
  \begin{resultbox}[Approximation 1: TM direction replacement]
    \vspace{-4pt}
    {\small Replace refracted TM direction by incident direction inside the medium.}
    \begin{itemize}\setlength\itemsep{0pt}
      \item {\small Error: $O\bigl(1/|\ntilde|^2\bigr) \le 4\,\%$ on cross-terms only}
      \item {\small Self-terms: exact (no approximation needed)}
    \end{itemize}
  \end{resultbox}
  \vspace{0pt}
  {\small \textbf{Physical reason:} for $|\ntilde| \gg 1$, Snell's law gives refracted angle
  $\theta_t \approx \sin\theta_i / |\ntilde| \ll 1$. All refracted rays propagate nearly
  normal to the surface.}
  \vspace{0pt}
  {\footnotesize At \SI{28}{\giga\hertz} with $|\ntilde| \approx 5.5$: correction $\le 3.3\,\%$ on cross-terms.}
  \vspace{-2pt}
  \returntoindex{p3-index}{Part III index}\quad\returntosummary{sum-p3-coherent-law}
\end{frame}

      \[
        \Gamma_{nn'} = \frac{\Lambda_{nn'}}{\sqrt{\Lambda_{nn}\,\Lambda_{n'n'}}}

% A-P3-8: Main theorem: coherent absorption law
% ----------------------------------------------------------
\begin{frame}[label=app-p3-theorem]{Main theorem: coherent absorption law}

  Under Approximations 1 and 2, the double sum collapses to a squared norm:

  \begin{resultbox}[Coherent absorption law]
    \vspace{-6pt}
    \[
      \Sab(\rr) = \bigl\|\Gtilde(\rr)\,\xx\bigr\|^2,
      \qquad
      \Gtilde(\rr) \in \Complex^{3 \times M}
    \]
    \vspace{-6pt}
  \end{resultbox}

  \vspace{2pt}
  \textbf{Column $j$:}\;
  $[\Gtilde(\rr)]_j = \sum_{n:\, j(n) = j}
    \sqrt{\Lambda_{nn}(\rr)}\; F_n(\rr)\,\bm{\psi}_n$

  \vspace{4pt}
  {\small
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Incoherent limit:}
      \begin{itemize}\setlength\itemsep{1pt}
        \item Phases of $\bm{\psi}_n$ random
        \item Cross-terms $\to 0$
        \item Reduces to Part I formula
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \textbf{Coherent peak:}
      \begin{itemize}\setlength\itemsep{1pt}
        \item All $N$ paths co-phase at $\rr$
        \item $\Sab \propto N^2$ (vs $N$ incoherent)
        \item Factor $N$ enhancement
        \item Precoder focuses energy on body
      \end{itemize}
    \end{column}
  \end{columns}
  }

  \vspace{4pt}
  {\footnotesize Approximation error $\le 5\,\%$ on cross-terms; self-terms exact.}

  \vspace{2pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% A-P3-9: Exposure operator properties
% ----------------------------------------------------------
\begin{frame}[label=app-p3-Q-props]{Exposure operator properties}

  \begin{resultbox}[Exposure operator]
    \vspace{-4pt}
    \[
      \QQ = \int_\Sigma \Gtilde(\rr)^H\,\Gtilde(\rr)\,\diff A
      \;\in\; \Complex^{M \times M}
    \]
  \end{resultbox}

  \vspace{2pt}
  \textbf{Properties:}
  Hermitian PSD ($\QQ = \QQ^H$, $\xx^H\QQ\,\xx \ge 0$).
  Rank $\le 3 \times |\text{illuminated surface pts}|$; full rank when $M \ll$ surface points.

  \vspace{2pt}
  \textbf{Eigendecomposition:} $\QQ = \sum_{k=1}^{M} \lambda_k\,\vv_k\,\vv_k^H$, \quad $\lambda_1 \ge \lambda_2 \ge \cdots \ge 0$

  \vspace{2pt}
  {\small
  \begin{tabular}{lll}
    \toprule
    Quantity & Meaning & Role \\
    \midrule
    $\vv_1$ & Dominant eigenvector & Worst-case precoder \\
    $\lambda_{\max}$ & Largest eigenvalue & Max absorption / unit TX power \\
    $\tr(\QQ)$ & Trace & Isotropic-TX expected absorption \\
    $\vv_k$, $\lambda_k$ & $k$-th eigenpair & $k$-th exposure mode \\
    \bottomrule
  \end{tabular}
  }

  \vspace{0pt}
  {\footnotesize Each $\vv_k$ defines a distinct spatial absorption pattern on the body surface.}
  \vspace{-4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% A-P3-10: Exposure-signal alignment derivation
% ----------------------------------------------------------
\begin{frame}[label=app-p3-rho]{Exposure-signal alignment $\rho$: derivation}

  \textbf{MRT precoder:} $\xx_{\mathrm{MRT}} = \sqrt{P}\;\hh^*/\|\hh\|$

  \vspace{4pt}
  Absorbed power under MRT:
  \[
    P_{\mathrm{abs}}^{\mathrm{MRT}}
    = \xx_{\mathrm{MRT}}^H \QQ\,\xx_{\mathrm{MRT}}
    = P\;\frac{\hh^\top \QQ\,\hh^*}{\|\hh\|^2}
  \]

  \begin{resultbox}[Exposure-signal alignment]
    \vspace{-4pt}
    \[
      \rho \;\equiv\; \frac{\hh^\top \QQ\,\hh^*}{\|\hh\|^2\;\lambda_{\max}(\QQ)}
      \;\in\; [0,\,1]
    \]
    \vspace{-4pt}
    {\small Bound follows from the Rayleigh quotient: $\hh^\top\QQ\,\hh^*/\|\hh\|^2 \le \lambda_{\max}(\QQ)$.}
  \end{resultbox}

  \vspace{2pt}
  {\small
  Three reasons signal focus $\neq$ body hotspot ($\rho < 1$):
  (1)~Polarisation mismatch body vs UE;
  (2)~Grazing paths: $F_n \to 0$ but still contribute to signal;
  (3)~Phase divergence between UE and body decorrelates $\hh$ from $\QQ$.
  }

  \vspace{0pt}
  {\footnotesize $\rho = 1$: MRT $\equiv$ worst-case precoder. \quad $\rho = 0$: MRT in null space of $\QQ$.}
  \vspace{-4pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}

% A-P3-11: QCQP formulation and solution
% ----------------------------------------------------------
\begin{frame}[label=app-p3-qcqp]{QCQP formulation and solution}

  {\small \textbf{Problem:} maximise signal power subject to absorption and power limits:}
  \vspace{-2pt}
  \[
    \max_{\xx}\; |\hh^\top\xx|^2
    \quad\text{s.t.}\quad
    \xx^H\QQ\,\xx \le P_{\mathrm{abs}}^{\max},
    \quad \|\xx\|^2 \le P
  \]
  \vspace{-2pt}
  {\small This is a QCQP. S-procedure: two quadratic constraints with PSD matrices $\Rightarrow$ strong duality.}
  \begin{resultbox}[Optimal precoder (KKT solution)]
    \vspace{-6pt}
    \[
      \xx^\star = \sqrt{P}\;
      \frac{(\lambda\,\QQ + \nu\,\II)^{-1}\,\hh^*}
           {\bigl\|(\lambda\,\QQ + \nu\,\II)^{-1}\,\hh^*\bigr\|}
    \]
    \vspace{-6pt}
  \end{resultbox}
  \vspace{-2pt}
  {\footnotesize $\lambda, \nu \ge 0$: dual variables found by bisection.}
  \vspace{0pt}
  {\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    Regime & Condition & Behaviour \\
    \midrule
    Unconstrained & $\lambda = 0$ & MRT: $\xx^\star \propto \hh^*$ \\
    Body-avoiding & $\lambda \gg \nu$ & Suppresses high-$\lambda_k$ eigenvectors of $\QQ$ \\
    Natural protection & $\rho \ll 1$ & MRT satisfies constraint; minimal signal loss \\
    \bottomrule
  \end{tabular}
  }
  \vspace{0pt}
  {\footnotesize Same structure as Ying \textit{et al.}\ 2015, but $\QQ$ derived from Fresnel physics, not FDTD-calibrated.}
  \vspace{-2pt}
  \returntoindex{p3-index}{Part III index}
\end{frame}
