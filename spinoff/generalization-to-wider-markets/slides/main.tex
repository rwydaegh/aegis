% ---------------------------------------------------------- 1. METHOD I
\begin{frame}{Dosimetry uses FDTD, I introduce physical optics}
  \vspace{2pt}
  \begin{center}
    {\large $\displaystyle
      \Sab(\rr) \;=\;
      {\color{colsinc}\Sinc}
      \;\cdot\; {\color{coltransmission}\Teff(\rr)}
      \;\cdot\; {\color{colgeometry}\ReLU\!\bigl[\nhat(\rr)\cdot(-\khat)\bigr]}
    $}
  \end{center}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}\setlength\itemsep{3pt}
        \item The body is \SI{1}{\metre} across, the skin depth at \SI{28}{\giga\hertz} is $\delta\approx\SI{0.3}{\milli\metre}$. FDTD needs ${\sim}10^{12}$ cells and days to weeks of GPU per run
        \item Above \SI{6}{\giga\hertz} the regulated quantity is absorbed power density on the skin, not whole-body SAR. Compliance is mandatory for every 5G/6G device and base station, at every product cycle
        \item So I integrate on the surface instead: {\color{colsinc}incident power} $\times$ {\color{coltransmission}Fresnel transmission into tissue} $\times$ {\color{colgeometry}an exact visibility gate}
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[height=4.1cm]{fig_geometry_sinc.pdf}
    \end{column}
  \end{columns}
  \vspace{1pt}
  \begin{resultbox}\centering
    One surface evaluation replaces a volume simulation: milliseconds instead of weeks, about $10^6$ faster than FDTD.
  \end{resultbox}
\end{frame}

% ---------------------------------------------------------- 2. METHOD II
\begin{frame}{Variable `fidelity levels' and differentiable}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering
      \includegraphics[height=3.15cm]{apd_angle_panel_T.pdf}\\[1pt]
      {\scriptsize $T_s$ falls, $T_p$ rises to the pseudo-Brewster peak, the unpolarized average stays flat at one scalar $T_0$.}
    \end{column}
    \begin{column}{0.56\textwidth}
      {\footnotesize One transmission model, at increasing fidelity:}
      \begin{itemize}\setlength\itemsep{1pt}\footnotesize
        \item scalar $T_0$: for tissue the polarization average collapses to one angle-independent number (proven here)
        \item full Fresnel $\Teff(\theta,\text{pol},f)$
        \item layered-media transfer matrix (covers, radomes)
        \item coherent MIMO: the field becomes a vector operator on the body
      \end{itemize}
      \vspace{1pt}
      {\footnotesize The surface computation is differentiable end to end, so placement, geometry, and precoding become gradient-based design and have ML applications.}
    \end{column}
  \end{columns}
  \vspace{0pt}
  \begin{resultbox}[The operator calculus]
    \vspace{-7pt}
    \[
      P_{\mathrm{abs}}=\xx^H\QQ\,\xx,\qquad
      \QQ=\int_\Sigma \Gtilde^H\Gtilde\,\diff A \ \ \text{(Hermitian PSD)},\qquad
      \xx^\star\propto(\lambda\QQ+\nu\II)^{-1}\hh^*
    \]
    \vspace{-9pt}

    {\scriptsize Incumbent PO/SBR solvers (Perceive EM, FEKO, HFSS SBR+) are already fast, but not differentiable or in this operator form.}
  \end{resultbox}
\end{frame}
