\begin{abstract}
The exposure-constrained beamformer (ECBF) of the AEGIS monograph
requires both the UE channel vector $\hh\in\C^M$ and the body exposure
operator $\Q\in\C^{M\times M}$. Standard pilot sounding delivers $\hh$
but not $\Q$. We give three sets of results. First, a structural
identifiability theorem: $\Q$ cannot be recovered from $\hh$ alone, and
no nontrivial $\hh$-only bound on the exposure-signal alignment $\rho$
exists across all admissible Hermitian PSD operators. Second, an
operator-theoretic perturbation framework that bounds $\|\Q(\xi)-\Q(\xi_0)\|_2$
by a single Lipschitz constant on the body-channel map $\Top_\xi$,
together with three control corollaries (absorbed power, robust
compliance margin, alignment drift). Third, a refinement of the
perturbation analysis that separates rigid translation from rotation:
translation invalidates $\Q$ on a sub-wavelength scale via a $k_0$
prefactor, while rotation is radian-scale; the implied refresh rule is
sharper than a single bulk constant suggests. We close with a
constructive $\hh$-witness for ECBF necessity that requires only modest
side information, the minimal hardware/protocol upgrade for direct
$\psib_n$ recovery, and a two-timescale ECBF architecture
$(\hh\!\sim\! T_c,\ \Q\!\sim\! T_p \gg T_c)$ with a clean staleness
margin.
\end{abstract}

% =================================================
\section{Setup and notation}
\label{sec:setup}

We adopt the conventions of Part~III of the AEGIS monograph.
A base station with $M$ antenna elements illuminates a scene containing
one body and one UE at $\rr_{\mathrm{UE}}$. Site-specific propagation
generates $N$ paths. Path $n$ has

\begin{itemize}[leftmargin=2em,itemsep=2pt]
\item arrival direction $\kh_n \in S^2$ at the body;
\item polarisation--amplitude vector $\psib_n \in \C^3$,
      $\psib_n \perp \kh_n$ (so $4$ real dof per path, encoded in the
      $(\hat{\boldsymbol\theta}(\kh_n),\hat{\boldsymbol\varphi}(\kh_n))$
      basis);
\item originating BS element index $j(n)\in\{1,\dots,M\}$.
\end{itemize}

The UE has receive antenna pattern $\Cp_R(\kh)\in\C^2$. The
\emph{path data} is
\[
  \mathcal{P} \;=\; \{(\psib_n,\kh_n,j(n))\}_{n=1}^N,
\]
with $6N$ continuous real dof plus $N$ discrete labels.

\paragraph{Channel projection.} The UE channel
$\hh = (h_1,\dots,h_M)^T \in \C^M$ has entries
\begin{equation}
  h_j = \sum_{n:\,j(n)=j}
        a_n\,e^{-ik_0\kh_n\cdot\rr_{\mathrm{UE}}},
  \qquad
  a_n = \Cp_R(\kh_n)^H\psib_n \in \C.
  \label{eq:h-def}
\end{equation}
Write $\pi_h:\mathcal{P}\to\C^M$ for this channel projection.

\paragraph{Exposure operator.}
For body surface $\Sigmaset$ with normal $\nh(\rr)$, the per-path
Fresnel transmission operator $\F_n(\rr):\C^3\to\C^3$ is supported on
$\mu_n(\rr)=-\kh_n\cdot\nh(\rr)>0$. The body-surface field channel is
\begin{equation}
  \Gt_j(\rr) = \sum_{n:\,j(n)=j}\F_n(\rr)\psib_n\,
               e^{-ik_0\kh_n\cdot\rr},
\end{equation}
and the exposure operator is the Gram integral
\begin{equation}
  \Q = \int_{\Sigmaset}\Gt(\rr)^H\Gt(\rr)\,\diff A
     \in\C^{M\times M},
  \qquad P_{\mathrm{abs}} = \xx^H\Q\xx.
  \label{eq:Q-def}
\end{equation}
Write $\pi_Q:\mathcal{P}\to\mathcal{H}_M^+$ for this operator
construction.

\paragraph{ECBF and alignment.}
The closed-form ECBF precoder is
\begin{equation}
  \xx^\star = \sqrt{P}\;
              \frac{(\lambda\Q+\nu\mathbf{I})^{-1}\hh^*}
                   {\norm{(\lambda\Q+\nu\mathbf{I})^{-1}\hh^*}},
  \qquad \lambda,\nu\ge 0\ \text{from KKT.}
  \label{eq:ecbf}
\end{equation}
The exposure-signal alignment under MRT is
\begin{equation}
  \rho \;=\; \frac{\hh^T\Q\hh^*}{\norm{\hh}^2\,\lammax(\Q)} \in [0,1].
  \label{eq:rho-def}
\end{equation}

\section{Identifiability of \texorpdfstring{$\Q$}{Q} from \texorpdfstring{$\hh$}{h}}
\label{sec:identifiability}

\subsection{Per-path information loss}

\begin{proposition}[Per-path information loss]
\label{prop:per-path-loss}
For each path, the channel projection retains $2$ of the $4$
continuous polarisation dof of $\psib_n$ and only the scalar phase
$e^{-ik_0\kh_n\cdot\rr_{\mathrm{UE}}}$ of the $2$ continuous arrival-angle
dof of $\kh_n$. The operator construction retains all $6$ continuous
dof of every path, and additionally couples them to body geometry over
a $2$-manifold of measurement points.
\end{proposition}

\begin{proof}
$a_n=\Cp_R(\kh_n)^H\psib_n$ is a rank-$1$ functional on $\C^2$, giving
$2$ real dof out of $4$. Two paths with distinct $\kh_n$ that share
$e^{-ik_0\kh_n\cdot\rr_{\mathrm{UE}}}$ are indistinguishable in $h_j$.
For $\pi_Q$: the per-path field
$\F_n(\rr)\psib_n e^{-ik_0\kh_n\cdot\rr}$ is rank-$2$ in $\psib_n$
($\F_n$ injective on its range) and depends on $\kh_n$ through both
$\mu_n(\rr)$ and the phase ramp over $\Sigmaset$.
\end{proof}

\subsection{Constructive non-identifiability}

\begin{proposition}[Generic non-identifiability]
\label{prop:non-id}
Whenever $N\ge 1$, the map $\pi_Q$ does not factor through $\pi_h$:
there exist $\mathcal{P},\mathcal{P}'$ with
$\pi_h(\mathcal{P})=\pi_h(\mathcal{P}')$ but
$\pi_Q(\mathcal{P})\ne\pi_Q(\mathcal{P}')$.
\end{proposition}

\begin{proof}
Pick any path $n$. Replace $\psib_n$ by $\psib_n+\boldsymbol\eta_n$ where
$\boldsymbol\eta_n\in\C^3$ is non-zero, satisfies
$\boldsymbol\eta_n\perp\kh_n$, and lies in
$\ker\Cp_R(\kh_n)^H$ (one-complex-dimensional, hence non-trivial).
Then $a_n$ is unchanged so $\hh$ is unchanged, but
$\F_n\boldsymbol\eta_n\not\equiv 0$ generically, so $\Gt$ and $\Q$
change. The kernel direction is the cross-polarisation component
absorbed by the body but invisible to the UE antenna.
\end{proof}

\subsection{No nontrivial \texorpdfstring{$\hh$}{h}-only bound on \texorpdfstring{$\rho$}{rho}}

The natural follow-up question is whether the BS can bound $\rho$
from $\hh$ alone (which would let it decide ECBF is unnecessary
without estimating $\Q$). The answer is sharply negative across the
full PSD class.

\begin{proposition}[No nontrivial $\hh$-only bound on $\rho$]
\label{prop:no-rho-bound}
Fix any nonzero $\hh\in\C^M$ and any target $\alpha\in[0,1]$. There
exists a Hermitian PSD matrix $\Q$ such that $\rho(\hh,\Q)=\alpha$.
\end{proposition}

\begin{proof}
Let $\mathbf{u}=\hh^*/\norm{\hh}$ and $\mathbf{v}\perp\mathbf{u}$ a unit
vector. For any $\lambda>0$, set
$\Q_\alpha = \lambda(\alpha\,\mathbf{u}\mathbf{u}^H+\mathbf{v}\mathbf{v}^H)$.
Then $\Q_\alpha$ is PSD, $\lammax(\Q_\alpha)=\lambda$, and
$\hh^T\Q_\alpha\hh^*/\norm{\hh}^2 = \mathbf{u}^H\Q_\alpha\mathbf{u}=\alpha\lambda$,
so $\rho(\hh,\Q_\alpha)=\alpha$.
\end{proof}

\begin{remark}
\label{rem:scope}
Prop.~\ref{prop:no-rho-bound} ranges over all Hermitian PSD matrices.
Restricted to the (much smaller) class of physically realisable $\Q$
generated by AEGIS-style path data on a geometrically constrained body,
useful $\hh$-based bounds do exist provided the BS additionally has
arrival-direction estimates and a coarse body-incidence support;
see \S\ref{sec:rho-witness}.
\end{remark}

% =================================================
\section{Three recovery routes}
\label{sec:routes}

\subsection{Route A: ray tracing on a digital twin}

If the BS holds a scene model $\mathcal{S}$ (walls, scatterers, body
pose), a differentiable ray tracer $\mathcal{R}:\mathcal{S}\to\mathcal{P}$
delivers path data and $\Q=\pi_Q(\mathcal{R}(\mathcal{S}),\Sigmaset)$.
This is the AEGIS pipeline (DiffeRT integration, level-7 and
level-8 kernels). Sufficiency is by construction; the operational error
is driven by mismatch between $\mathcal{S}$ and ground truth, which we
quantify in \S\ref{sec:perturbation}.

\subsection{Route B: minimum pilot diversity for direct \texorpdfstring{$\psib_n$}{psi\_n} recovery}

\begin{proposition}[Minimal pilot diversity for $\psib_n$]
\label{prop:pilot-diversity}
Suppose the UE has a dual-polarised receive front end with two
linearly independent patterns $\Cp_R^{(1)}(\kh),\Cp_R^{(2)}(\kh)\in\C^2$
(rank-$2$ port matrix), and the per-path arrival direction $\kh_n$ is
known (e.g.\ from UE-side AoA estimation on a small array). Then
$\psib_n$ is identifiable from the per-path port amplitudes
$a_n^{(p)}=\Cp_R^{(p)}(\kh_n)^H\psib_n$, $p=1,2$.
\end{proposition}

\begin{proof}
With $\kh_n$ known, $\psib_n$ lives in the $2$-complex-dimensional plane
$\kh_n^\perp\subset\C^3$. The two scalar measurements form a $2\times 2$
linear system in the
$(\hat{\boldsymbol\theta}(\kh_n),\hat{\boldsymbol\varphi}(\kh_n))$ basis;
linear independence of $\Cp_R^{(1)},\Cp_R^{(2)}$ guarantees invertibility.
\end{proof}

Route B reduces $\Q$-acquisition to Route A with measurement-derived
$\mathcal{P}$ in place of ray-traced $\mathcal{P}$ (the body geometry is
still required to integrate $\Gt^H\Gt$ over $\Sigmaset$). Hardware cost:
dual-pol UE Rx (already standard at FR2), UE-side AoA capability
(already used for spatial multiplexing), and a protocol extension that
exposes per-port per-path amplitudes to the BS.

\subsection{Route C: phantom-calibrated envelope}

A phantom-calibrated $\Q^{\mathrm{phantom}}$ in a fixed canonical pose
yields a one-shot static bound
$P_{\mathrm{abs}} \le \lammax(\Q^{\mathrm{phantom}})\,P$ for every
$\norm{\xx}^2\le P$. This is the AEGIS analogue of Hochwald
\textit{et~al.}'s phantom-fitted SAR matrix $\mathbf{S}_V$. Useful as a
fallback envelope and as an h-only safety floor, but loose because it
must absorb every body pose that could arise during the certification
window.

% =================================================
\section{Operator-theoretic perturbation of \texorpdfstring{$\Q$}{Q}}
\label{sec:perturbation}

We now bound $\Q(\xi)-\Q(\xi_0)$ for body pose
$\xi=(R,\mathbf{t})\in SE(3)$. The path data outside the body is fixed.

\subsection{Operator framework and a generic Lipschitz bound}
\label{sec:op-framework}

\begin{definition}[Body-channel map]
For each pose $\xi$, define the bounded linear operator
\[
  \Top_\xi:\C^M \to L^2(\Sigmaset_0;\C^3),
  \qquad
  (\Top_\xi\xx)(\rr) = \Gt_\xi(\rr)\,\xx,
\]
where $\Sigmaset_0$ is a fixed reference surface and the
pose-transformed integrand is pulled back to $\Sigmaset_0$ via the
rigid map $g_\xi$. Then
\[
  \Q(\xi) = \Top_\xi^*\Top_\xi.
\]
\end{definition}

\begin{proposition}[Operator-norm perturbation]
\label{prop:op-perturbation}
For any two poses $\xi,\xi_0$,
\begin{equation}
  \norm{\Q(\xi)-\Q(\xi_0)}_2 \;\le\;
  \bigl(\norm{\Top_\xi}_2 + \norm{\Top_{\xi_0}}_2\bigr)\,
  \norm{\Top_\xi-\Top_{\xi_0}}_2.
  \label{eq:op-norm-bound}
\end{equation}
\end{proposition}

\begin{proof}
$\Top_\xi^*\Top_\xi - \Top_{\xi_0}^*\Top_{\xi_0}
 = (\Top_\xi - \Top_{\xi_0})^*\Top_\xi + \Top_{\xi_0}^*(\Top_\xi - \Top_{\xi_0})$;
apply triangle inequality and submultiplicativity.
\end{proof}

\begin{corollary}[Lipschitz drift under smooth motion]
\label{cor:lipschitz}
If locally $\norm{\Top_\xi-\Top_{\xi_0}}_2 \le L_T\norm{\xi-\xi_0}$ and
$\sup_\zeta\norm{\Top_\zeta}_2\le B$, then
\[
  \norm{\Q(\xi)-\Q(\xi_0)}_2 \le 2BL_T\norm{\xi-\xi_0}
  \;\equiv\; L_Q\norm{\xi-\xi_0}.
\]
\end{corollary}

This is the abstract bound. It is correct but it bundles two
qualitatively different regimes; we now separate them.

\subsection{Translation: sub-wavelength sensitivity (the \texorpdfstring{$k_0$}{k0} prefactor)}

\begin{proposition}[Translation perturbation]
\label{prop:translation}
Under rigid translation $\rr\mapsto\rr+\mathbf{t}$ with $\mathbf{t}\in\R^3$,
\begin{equation}
  [\Q(\mathbf{t})]_{ab} =
  \sum_{\substack{n:\,j(n)=a\\n':\,j(n')=b}}
  e^{-ik_0(\kh_n-\kh_{n'})\cdot\mathbf{t}}\,Q^{(nn')}_{ab},
  \label{eq:Q-trans}
\end{equation}
where $Q^{(nn')}_{ab}$ is the path-pair contribution to $[\Q(0)]_{ab}$.
For $k_0\norm{\mathbf{t}}\ll 1$,
\begin{equation}
  \frac{\norm{\Q(\mathbf{t})-\Q(0)}_F}{\norm{\Q(0)}_F}
  \le 2 k_0\norm{\mathbf{t}} + O\bigl((k_0\norm{\mathbf{t}})^2\bigr).
  \label{eq:Q-trans-bound}
\end{equation}
\end{proposition}

\begin{proof}[Sketch]
Translation leaves $\nh$, $\mu_n$, $\F_n$ invariant; each per-path field
acquires a phasor $e^{-ik_0\kh_n\cdot\mathbf{t}}$ that factors out of
the area integral and survives in the $\Gt^H\Gt$ quadratic as
$e^{-ik_0(\kh_n-\kh_{n'})\cdot\mathbf{t}}$. Linearise and bound by sum
of moduli.
\end{proof}

\paragraph{Operational reading.}
At $28$\,GHz, $\lambda_0\!\approx\!1.07$\,cm, so $k_0\!\approx\!587$\,m$^{-1}$.
Translation by $\norm{\mathbf{t}}=1$\,mm gives
$k_0\norm{\mathbf{t}}\!\approx\!0.59$\,rad, sufficient to move $\Q$ by
more than its own Frobenius norm. Translation invalidates $\Q$ on a
sub-wavelength scale. In the language of
Cor.~\ref{cor:lipschitz}, the bulk Lipschitz constant for a translation
generator is $L_Q^{\mathrm{trans}}\!=\!\Theta(k_0)$.

\subsection{Rotation: radian sensitivity (no \texorpdfstring{$k_0$}{k0} prefactor on the smooth term)}

\begin{proposition}[Rotation perturbation]
\label{prop:rotation}
Under rigid rotation by angle $\theta$ about $\hat{\boldsymbol\omega}\in S^2$,
assuming the visible-path set is unchanged,
\begin{equation}
  \frac{\norm{\Q(\theta\hat{\boldsymbol\omega})-\Q(0)}_F}{\norm{\Q(0)}_F}
  \le C_{\mathrm{rot}}^{\mathrm{smooth}}\,\theta
    + C_{\mathrm{rot}}^{\mathrm{phase}}\,k_0\,\mathrm{diam}(\Sigmaset)\,\theta
    + O(\theta^2),
\end{equation}
where $C_{\mathrm{rot}}^{\mathrm{smooth}}$ depends on the maximum slope
of Fresnel coefficients $|\partial_\mu t_{s,n}|,|\partial_\mu t_{p,n}|$
on visible paths, and $C_{\mathrm{rot}}^{\mathrm{phase}}$ is a
$O(1)$ geometric constant.
\end{proposition}

\begin{proof}[Sketch]
Rotation perturbs $\nh(\rr)$ by $\theta\hat{\boldsymbol\omega}\times\nh(\rr)$,
giving $\delta\mu_n=O(\theta)$; Fresnel amplitudes are smooth in $\mu>0$,
so the smooth term is $O(\theta)$ and $k_0$-free. The TE/TM bases rotate
at the same rate. The phase ramp acquires a per-point variation
$e^{-ik_0\kh_n\cdot\delta\rr(\theta)}$ with $\norm{\delta\rr}=O(\theta\norm{\rr})$,
which contributes the body-diameter-scaled phase term.
\end{proof}

\paragraph{Operational reading.}
For a $50$\,cm body at $28$\,GHz the phase term is $\sim\!300\theta$,
so beyond $\theta\!\sim\!0.003$\,rad ($\sim\!0.2^\circ$) interference
patterns far from the rotation axis decohere. The smooth term remains
$\sim\!\theta$. Both are radian-scale: rotation by $\theta\!\sim\!0.1$\,rad
($\sim\!6^\circ$) typically perturbs $\Q$ by $\sim\!10\%$ in the
near-axis region and substantially more in the far-axis region.

\subsection{Self-shadowing transitions}
\label{sec:self-shadow}

The smooth bounds above assume the visible-path set is unchanged. A
path with $\mu_n(\rr;0)=0$ at some $\rr$ enters or leaves the visible
set under arbitrarily small pose perturbation, contributing
discontinuously to $\Q$. These transitions are measure-zero in pose
space (a finite union of codimension-$1$ strata) but their cumulative
contribution over a finite pose excursion is bounded by the silhouette
line integral
\begin{equation}
  \norm{\delta\Q^{\mathrm{shadow}}}_F \le
  \int_{\partial\Sigmaset_{\mathrm{vis}}}
  \norm{\Gt^{(n)}(\rr)}^2\,\diff\ell.
\end{equation}
Generically small in ECBF-relevant scenarios because the dominant
exposure modes correspond to near-normal incidence, away from the
silhouette. Cor.~\ref{cor:lipschitz}'s smoothness assumption fails
exactly here, and any practical $L_Q$ estimate must cap shadow
transitions separately.

\subsection{Three control corollaries}
\label{sec:control-corollaries}

The operator-norm perturbation
$\varepsilon_Q\equiv\norm{\Q-\Qhat}_2$ controls everything we need for
robust ECBF design.

\begin{corollary}[Uniform absorbed-power perturbation]
\label{cor:uniform-power}
For any $\xx$ with $\norm{\xx}^2\le P$,
$\abs{\xx^H\Q\xx-\xx^H\Qhat\xx} \le P\varepsilon_Q$.
\end{corollary}

\begin{corollary}[Robust compliance margin]
\label{cor:robust-margin}
A precoder designed to satisfy
$\xx^H\Qhat\xx \le P_{\mathrm{abs}}^{\max} - P\varepsilon_Q$
is guaranteed to satisfy $\xx^H\Q\xx \le P_{\mathrm{abs}}^{\max}$ under
true $\Q$.
\end{corollary}

\begin{corollary}[Worst-case eigenvalue drift]
$\abs{\lammax(\Q)-\lammax(\Qhat)}\le\varepsilon_Q$ (Weyl).
\end{corollary}

\begin{corollary}[Alignment drift]
\label{cor:rho-drift}
With $\mu\equiv\min\{\lammax(\Q),\lammax(\Qhat)\}>0$,
$\abs{\rho(\hh,\Q)-\rho(\hh,\Qhat)}\le 2\varepsilon_Q/\mu$.
\end{corollary}

\begin{proof}[of Cor.~\ref{cor:rho-drift}]
Write $a=\mathbf{u}^H\Q\mathbf{u}$, $\widehat a=\mathbf{u}^H\Qhat\mathbf{u}$,
$b=\lammax(\Q)$, $\widehat b=\lammax(\Qhat)$ with
$\mathbf{u}=\hh^*/\norm{\hh}$. Then
$\abs{a-\widehat a}\le\varepsilon_Q$ and
$\abs{b-\widehat b}\le\varepsilon_Q$, so
$\abs{a/b-\widehat a/\widehat b}
 \le\abs{a-\widehat a}/b + \widehat a\,\abs{1/b-1/\widehat b}
 \le\varepsilon_Q/\mu + \widehat b\,\varepsilon_Q/(b\widehat b)
 \le 2\varepsilon_Q/\mu$.
\end{proof}

\section{Two-timescale ECBF and refresh interval}
\label{sec:timescales}

Let $T_c$ be the channel coherence time (typical $1$--$10$\,ms at FR2)
and $T_p$ the body-pose coherence interval (typical $0.1$--$1$\,s).

\begin{definition}[Two-timescale ECBF]
\label{def:two-timescale}
Pilot sounding refreshes $\hh(t)$ every slot $\le T_c$; a separate
body-pose channel refreshes $\Qhat(t_p)$ every interval $\le T_p$. The
per-slot precoder is
\[
  \xx^\star(t) = \sqrt{P}\,
  \frac{(\lambda(t)\Qhat(t_p^-) + \nu(t)\mathbf{I})^{-1}\hh(t)^*}{\norm{\cdot}},
\]
with $(\lambda(t),\nu(t))$ re-solved per slot from
$(\hh(t),\Qhat(t_p^-))$.
\end{definition}

\begin{corollary}[Safe refresh interval]
\label{cor:refresh}
If body pose evolves with bounded speed $\norm{\dot\xi(t)}\le v_{\max}$
and $\Q(\xi)$ has local Lipschitz constant $L_Q$ along the actual
motion, then over an update interval $\Delta t$,
\[
  \norm{\Q(t+\Delta t)-\Q(t)}_2 \le L_Q v_{\max}\Delta t.
\]
To guarantee operator error budget $\varepsilon_Q$ it is sufficient to
refresh at
\begin{equation}
  \Delta t_{\max} = \frac{\varepsilon_Q}{L_Q v_{\max}}.
  \label{eq:dt-max}
\end{equation}
\end{corollary}

\paragraph{Refining $L_Q$ via the translation--rotation split.}
Cor.~\ref{cor:refresh} writes a single $L_Q$ but
Props.~\ref{prop:translation}--\ref{prop:rotation} show this hides a
factor of $\sim\!k_0$ between regimes. Decomposing $\dot\xi=(\dot{\mathbf{t}},\dot{\boldsymbol\omega})$
with $\norm{\dot{\mathbf{t}}}\le v_{\max}^{\mathrm{trans}}$ and
$\norm{\dot{\boldsymbol\omega}}\le \omega_{\max}^{\mathrm{rot}}$,
\begin{equation}
  \Delta t_{\max} \;\approx\;
  \frac{\varepsilon_Q\,\norm{\Q}_F^{-1}}
       {2 k_0 v_{\max}^{\mathrm{trans}}
        + \bigl(C_{\mathrm{rot}}^{\mathrm{smooth}}+
                C_{\mathrm{rot}}^{\mathrm{phase}} k_0\,\mathrm{diam}(\Sigmaset)\bigr)
        \omega_{\max}^{\mathrm{rot}}}.
  \label{eq:dt-refined}
\end{equation}
For typical mobile users the translation term dominates: a walking
user at $1.4$\,m/s gives $k_0 v_{\max}^{\mathrm{trans}}\!\sim\!820$\,rad/s,
forcing $\Delta t_{\max} \!\sim\! \varepsilon_Q / 820$\,s. To allow
$\varepsilon_Q\!=\!0.1\,\norm{\Q}_F$ with a walking user requires
refreshing $\Q$ every $\sim\!120$\,$\mu$s --- close to the channel
coherence time. The two-timescale architecture is therefore most
attractive for stationary or slow-moving users (seated, standing,
fixed-location IoT), where translation is negligible and only rotation
contributes.

% =================================================
\section{A constructive \texorpdfstring{$\hh$}{h}-witness for ECBF necessity}
\label{sec:rho-witness}

Prop.~\ref{prop:no-rho-bound} rules out a useful $\hh$-only bound on
$\rho$ across the unrestricted PSD cone. Restricting to physically
admissible $\Q$ (whose support is the body-incident path manifold)
recovers a useful bound when modest side information is available.

\begin{proposition}[$\hh$-witness with arrival-direction side information]
\label{prop:rho-witness}
Suppose the BS has access to (i) the channel $\hh$, (ii) the per-path
\emph{departure} directions $\{\kh_{T,n}\}_{n=1}^N$ at the BS array
(recoverable via spatial covariance + MUSIC/ESPRIT in the
multi-snapshot regime), and (iii) a coarse departure cone
$K\subset S^2$ that contains all directions whose corresponding paths
illuminate the body. Decompose
\[
  \hh = \hh_K + \hh_{K^c},
  \qquad
  \hh_K = \sum_{n:\,\kh_{T,n}\in K}
          a_n e^{-ik_0\kh_n\cdot\rr_{\mathrm{UE}}}\mathbf{e}_{j(n)}.
\]
Then
\begin{equation}
  \rho \;\le\; \frac{\norm{\hh_K}^2}{\norm{\hh}^2}.
  \label{eq:rho-witness}
\end{equation}
\end{proposition}

\begin{proof}
For physical $\Q$ generated by AEGIS-style path data,
$\Q\hh_{K^c}^*=\mathbf{0}$: every path contributing to $\Gt$ has
departure in $K$ by hypothesis. Hence $\hh^T\Q\hh^*=\hh_K^T\Q\hh_K^*$.
By Rayleigh--Ritz on PSD $\Q$,
$\hh_K^T\Q\hh_K^*\le\lammax(\Q)\norm{\hh_K}^2$. Divide by
$\norm{\hh}^2\lammax(\Q)$.
\end{proof}

\begin{remark}[Reconciliation with Prop.~\ref{prop:no-rho-bound}]
\label{rem:reconcile}
Prop.~\ref{prop:no-rho-bound}'s counterexample $\Q_\alpha$ requires
$\hh^*$ to lie in $\mathrm{range}(\Q_\alpha)$ with
$\mathbf{u}^H\Q_\alpha\mathbf{u}=\alpha\lambda$. For physically generated
$\Q$, $\mathrm{range}(\Q)$ is constrained to the body-incident BS
subspace; the choice $\Q_\alpha=\lambda(\alpha\mathbf{u}\mathbf{u}^H+\mathbf{v}\mathbf{v}^H)$
is admissible only if $\mathbf{u}=\hh^*/\norm{\hh}$ lies in this
subspace, which is precisely the setting in which Prop.~\ref{prop:rho-witness}
gives $\norm{\hh_K}^2/\norm{\hh}^2\!\to\!1$ and the witness becomes
trivial. The two propositions describe the same line from opposite
sides: when the channel is dominated by body-incident paths, $\rho$ can
be anything in $[0,1]$ and $\hh$ alone tells you nothing more than
``ECBF might matter''; when the channel is dominated by
non-body-incident paths, $\rho$ is small and ECBF is unnecessary.
\end{remark}

\paragraph{What the witness does \emph{not} say.}
Large $\norm{\hh_K}^2/\norm{\hh}^2$ does not imply $\rho$ is large.
The body might still be illuminated mostly along low-Fresnel-coupling
directions, or the dominant $\Q$-eigenmode might be misaligned with
the channel within the body-incident subspace. The witness is a
necessary condition for ECBF to matter, not sufficient. Sufficiency
requires $\Q$.

\paragraph{Combined safety certificate.}
Pair Prop.~\ref{prop:rho-witness} with Route C: if
$\lammax(\Q)\le\Lambda$ from a phantom envelope and the witness gives
$\norm{\hh_K}^2/\norm{\hh}^2\le\epsilon$, then
$P_{\mathrm{abs}}^{\mathrm{MRT}}\le P\epsilon\Lambda$, an
$\hh$-only-plus-static-envelope safety certificate that runs MRT
without ECBF whenever $\epsilon\Lambda\le P_{\mathrm{abs}}^{\max}/P$.

\section{What the JSAC SI submission can claim}
\label{sec:writeup}

\subsection{Closed-form claims (this note alone)}

\begin{enumerate}[leftmargin=2em,itemsep=2pt]
\item \textbf{Structural unidentifiability of $\Q$ from $\hh$}
      (Props.~\ref{prop:per-path-loss}--\ref{prop:non-id}). The UE
      antenna is a polarisation filter; the body is not.
\item \textbf{No nontrivial $\hh$-only bound on $\rho$}
      (Prop.~\ref{prop:no-rho-bound}). Constructive counterexample
      across the full PSD class.
\item \textbf{Minimum pilot diversity for direct $\psib_n$ recovery}
      (Prop.~\ref{prop:pilot-diversity}). Identifies the exact
      hardware/protocol upgrade.
\item \textbf{Operator-theoretic perturbation framework}
      (Prop.~\ref{prop:op-perturbation}, Cor.~\ref{cor:lipschitz}).
\item \textbf{Translation/rotation split}
      (Props.~\ref{prop:translation}--\ref{prop:rotation}). Shows the
      bulk Lipschitz constant hides a $k_0$ prefactor; translation
      dominates for mobile users.
\item \textbf{Self-shadowing line integral bound} (\S\ref{sec:self-shadow}).
\item \textbf{Three control corollaries}
      (Cors.~\ref{cor:uniform-power}--\ref{cor:rho-drift}). Robust
      compliance margin, eigenvalue drift, alignment drift.
\item \textbf{Two-timescale ECBF with refresh rule}
      (Def.~\ref{def:two-timescale}, Cor.~\ref{cor:refresh},
      \eqref{eq:dt-refined}).
\item \textbf{Constructive $\hh$-witness with side info}
      (Prop.~\ref{prop:rho-witness}). Reconciled with the no-go theorem
      in Remark~\ref{rem:reconcile}.
\end{enumerate}

\subsection{Empirical follow-ups (need AEGIS runs)}

All items below are runnable inside the existing AEGIS codebase using
the modules listed in Appendix~\ref{app:aegis-map}. The minimal scenario
is: one phantom (e.g.\ \texttt{thelonious}), one BS array, $28$\,GHz,
single-bounce or DiffeRT-traced multipath, one UE.

\begin{enumerate}[leftmargin=2em,itemsep=2pt]
\setcounter{enumi}{9}
\item \textbf{Tightness of perturbation bounds.}
      Compute $\norm{\Q(g)-\Q(0)}_2/\norm{\Q(0)}_2$ and the
      eigenvalue drift on a sweep of body translations
      $\{0,0.1,0.5,1,5,10\}$\,mm and rotations
      $\{0,1,5,10,30\}^\circ$. Compare to
      Props.~\ref{prop:translation}--\ref{prop:rotation}. Hypothesised
      result: Frobenius bounds tight to within a constant; spectral
      bounds have order-of-magnitude slack (Weyl is loose for
      structured perturbations).
\item \textbf{Empirical Lipschitz constants $L_Q^{\mathrm{trans}},L_Q^{\mathrm{rot}}$.}
      Finite differences over the sweep above. Substitute into
      \eqref{eq:dt-refined} and tabulate $\Delta t_{\max}$ for stationary,
      seated-fidgeting, walking, and jogging users.
\item \textbf{Witness sharpness.}
      For a sweep over UE positions and body locations, compare the
      witness $\norm{\hh_K}^2/\norm{\hh}^2$ to the true $\rho$. Identify
      the regime where the gap is below $1$\,dB (witness informative)
      versus above (witness trivial; $\Q$ truly required).
\item \textbf{Two-timescale ECBF in simulation.}
      Implement Def.~\ref{def:two-timescale} on a synthetic body-motion
      trace; compare end-to-end exposure compliance and spectral
      efficiency to (a) per-slot oracle ECBF, (b) static
      Hochwald-style worst-case ECBF. The robust margin from
      Cor.~\ref{cor:robust-margin} closes the safety gap.
\end{enumerate}

\subsection{Suggested paper structure for the JSAC SI}

Lead with the identifiability theorems (negative + Route-B positive),
then the perturbation framework (operator-theoretic + translation/rotation
split + control corollaries), then the two-timescale architecture as
the algorithmic contribution, then AEGIS as the digital-twin substrate
that realises Route A at per-slot cadence and instantiates every
quantity in the empirical sweep. The body of the paper closes the loop
between the identifiability gap (theory) and the AEGIS-enabled
robust-margin ECBF (system). The story sits squarely in the
``twin-in-the-loop'' / ``closed-loop optimisation'' theme of the call.
